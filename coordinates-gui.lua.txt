local function Create(class, properties, parent)
    local object = Instance.new(class)
    for property, value in pairs(properties) do
        object[property] = value
    end
    object.Parent = parent
    return object
end

local function Round(object)
    Create("UICorner", {
        CornerRadius = UDim.new(0, 6)
    }, object)
end

local function Stroke(object)
    Create("UIStroke", {
        Color = Green,
        Thickness = 1.5
    }, object)
end

local function GetRoot()
    local character = Player.Character
    return character and character:FindFirstChild("HumanoidRootPart")
end

local function SaveLocations()
    if type(writefile) ~= "function" then
        return false
    end

    local success = pcall(function()
        writefile(Config.LocationFile, HttpService:JSONEncode({
            tabs = Tabs,
            locations = Locations
        }))
    end)

    return success
end

local function LoadLocations()
    if type(isfile) ~= "function" or type(readfile) ~= "function" then
        return
    end

    local success, data = pcall(function()
        if isfile(Config.LocationFile) then
            return HttpService:JSONDecode(readfile(Config.LocationFile))
        end
    end)

    if success and type(data) == "table" then
        if type(data.tabs) == "table" and type(data.locations) == "table" then
            Tabs = data.tabs
            Locations = data.locations
            -- Remove the old built-in Default tab while keeping its saved locations.
            local defaultLocations = Locations.Default
            for index = #Tabs, 1, -1 do
                if Tabs[index] == "Default" then
                    table.remove(Tabs, index)
                end
            end
            if type(defaultLocations) == "table" and #defaultLocations > 0 then
                local migratedName = "Tab 1"
                local suffix = 1
                while table.find(Tabs, migratedName) do
                    suffix += 1
                    migratedName = "Tab " .. suffix
                end
                table.insert(Tabs, 1, migratedName)
                Locations[migratedName] = defaultLocations
            end
            Locations.Default = nil
            ActiveTab = Tabs[1]
        else
            -- Migrate the original single-list format into a normal, renameable tab.
            if #data > 0 then
                Tabs = {"Tab 1"}
                Locations = {["Tab 1"] = data}
                ActiveTab = "Tab 1"
            end
        end
    end
end
LoadLocations()

local function MakeDraggable(frame, dragArea)
    local dragging = false
    local dragStart
    local startPosition

    dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPosition = frame.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart

            frame.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
end

local function VerifyKey(key)
    local requestFunction = request or http_request

    if type(requestFunction) ~= "function" then
        return false, "HTTP requests unavailable"
    end

    local success, response = pcall(function()
        return requestFunction({
            Url = Config.VerificationEndpoint,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json"
            },
            Body = HttpService:JSONEncode({
                key = key
            })
        })
    end)

    if not success or type(response) ~= "table" then
        return false, "Connection failed"
    end

    if response.StatusCode and response.StatusCode ~= 200 then
        return false, "Server error"
    end

    local decoded
    local decodeSuccess = pcall(function()
        decoded = HttpService:JSONDecode(response.Body)
    end)

    if not decodeSuccess or type(decoded) ~= "table" then
        return false, "Invalid server response"
    end

    if decoded.valid == true then
        return true, "Verified"
    end

    return false, decoded.message or "Invalid or expired key"
end

    local function MoveTabIndicator(tabButton)
        -- Keep the indicator in the tab strip's scrolling content,
        -- not in MainFrame screen coordinates.
        local x = tabButton.AbsolutePosition.X - TabBar.AbsolutePosition.X
            + TabBar.CanvasPosition.X
        local width = tabButton.AbsoluteSize.X

        game:GetService("TweenService"):Create(
            TabIndicator,
            TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {
                Position = UDim2.fromOffset(x, 31),
                Size = UDim2.fromOffset(width, 3)
            }
        ):Play()
    end

    local function RefreshTabs()
        TabIndicator.Visible = ActiveTab ~= nil
        for _, child in ipairs(TabBar:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end

        for index, tabName in ipairs(Tabs) do
            local selected = tabName == ActiveTab
            local tabButton = Create("TextButton", {
                Size = UDim2.fromOffset(
                    math.clamp(#tostring(tabName) * 9 + 24, 76, 145), 30
                ),
                BackgroundColor3 = selected
                    and Color3.fromRGB(25, 65, 40) or Dark,
                Text = tostring(tabName),
                TextColor3 = selected and Green
                    or Color3.fromRGB(175, 175, 175),
                TextSize = 12,
                Font = Enum.Font.Code,
                LayoutOrder = index
            }, TabBar)
            Round(tabButton)

            tabButton.MouseButton1Click:Connect(function()
                ActiveTab = tabName
                RefreshTabs()
                RefreshLocations()
                task.defer(function()
                    for _, button in ipairs(TabBar:GetChildren()) do
                        if button:IsA("TextButton") and button.Text == ActiveTab then
                            MoveTabIndicator(button)
                            break
                        end
                    end
                end)
            end)

            -- Right-click a tab to rename it.
            tabButton.MouseButton2Click:Connect(function()
                local editor = Create("TextBox", {
                    Size = tabButton.Size,
                    Position = tabButton.Position,
                    BackgroundColor3 = Dark,
                    TextColor3 = Green,
                    PlaceholderText = "Tab name",
                    Text = tostring(tabName),
                    TextSize = 12,
                    Font = Enum.Font.Code,
                    ClearTextOnFocus = false,
                    ZIndex = tabButton.ZIndex + 2
                }, TabBar)
                Round(editor)
                tabButton.Visible = false
                editor:CaptureFocus()
                editor.FocusLost:Connect(function(enterPressed)
                    local newName = editor.Text:match("^%s*(.-)%s*$")
                    if enterPressed and newName ~= "" and #newName <= 24 then
                        local duplicate = false
                        for _, existing in ipairs(Tabs) do
                            if existing ~= tabName and string.lower(existing) == string.lower(newName) then
                                duplicate = true
                                break
                            end
                        end
                        if not duplicate then
                            Locations[newName] = Locations[tabName] or {}
                            Locations[tabName] = nil
                            for i, existing in ipairs(Tabs) do
                                if existing == tabName then Tabs[i] = newName break end
                            end
                            if ActiveTab == tabName then ActiveTab = newName end
                            SaveLocations()
                        end
                    end
                    editor:Destroy()
                    RefreshTabs()
                    RefreshLocations()
                end)
            end)

            if selected then
                task.defer(function()
                    if tabButton.Parent then
                        MoveTabIndicator(tabButton)
                    end
                end)
            end
        end
    end

    AddTabButton.MouseButton1Click:Connect(function()
        local number = 1
        local newName = "Tab " .. number

        while table.find(Tabs, newName) do
            number += 1
            newName = "Tab " .. number
        end

        table.insert(Tabs, newName)
        Locations[newName] = {}
        ActiveTab = newName

        SaveLocations()
        RefreshTabs()
        RefreshLocations()

        task.defer(function()
            for _, child in ipairs(TabBar:GetChildren()) do
                if child:IsA("TextButton") and child.Text == ActiveTab then
                    TabBar.CanvasPosition = Vector2.new(
                        math.max(0, child.AbsolutePosition.X - TabBar.AbsolutePosition.X),
                        0
                    )
                    MoveTabIndicator(child)
                    break
                end
            end
        end)
    end)

    SaveButton.MouseButton1Click:Connect(function()

        if not ActiveTab or not root or name == "" or #name > 30 then
            return
        end

        local activeLocations = GetActiveLocations()

        for _, location in ipairs(activeLocations) do
            if string.lower(location.name) == string.lower(name) then
                return
            end
        end

        local position = root.Position

        table.insert(activeLocations, {
            name = name,
            x = position.X,
            y = position.Y,
            z = position.Z
        })

        SaveName.Text = ""
        SaveLocations()
        RefreshLocations()
    end)

    RefreshTabs()
    RefreshLocations()

VerifyButton.MouseButton1Click:Connect(function()
    if KeyBox.Text == "" then
        Status.Text = "ENTER A KEY"
        return
    end

    VerifyButton.Text = "VERIFYING..."
    VerifyButton.Active = false
    Status.Text = ""

    local valid, message = VerifyKey(KeyBox.Text)

    if valid then
        Unlocked = true
        KeyFrame:Destroy()
        CreateMainGUI()
        ToggleButton.Visible = true
    else
        Status.Text = message
        VerifyButton.Text = "VERIFY KEY"
        VerifyButton.Active = true
    end
end)

            position.X,

        )
