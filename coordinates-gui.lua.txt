local function requestVerification(key)
    local requestFunction = request or http_request
    if not requestFunction then
        return false
    end

    local success, response = pcall(function()
        return requestFunction({
            Url = Config.VerificationEndpoint,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json"
            },
            Body = HttpService:JSONEncode({
                key = key
            })
        })
    end)

    if not success or not response or not response.Body then
        return false
    end

    local decodedSuccess, decoded = pcall(function()
        return HttpService:JSONDecode(response.Body)
    end)

    return decodedSuccess and decoded.valid == true
end

local function loadSavedKey()
    if not isfile or not readfile or not isfile(Config.SessionFile) then
        return nil
    end

    local success, data = pcall(function()
        return HttpService:JSONDecode(readfile(Config.SessionFile))
    end)

    if success and type(data) == "table" and type(data.key) == "string" then
        return data.key
    end

    return nil
end

local function saveKey(key)
    if writefile then
        pcall(function()
            writefile(Config.SessionFile, HttpService:JSONEncode({
                key = key
            }))
        end)
    end
end

local function deleteSavedKey()
    if delfile and isfile and isfile(Config.SessionFile) then
        pcall(function()
            delfile(Config.SessionFile)
        end)
    end
end

local function create(class, properties, parent)
    local object = Instance.new(class)
    for property, value in pairs(properties) do
        object[property] = value
    end
    object.Parent = parent
    return object
end

local function round(object, radius)
    create("UICorner", {
        CornerRadius = UDim.new(0, radius)
    }, object)
end

local function outline(object, color, transparency)
    create("UIStroke", {
        Color = color,
        Transparency = transparency or 0,
        Thickness = 1
    }, object)
end

local function makeDraggable(frame, handle)
    local dragging = false
    local dragStart
    local startPosition

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPosition = frame.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
end

local Tabs = {"Tab 1"}
local Locations = {}
local ActiveTab = "Tab 1"

local function loadLocations()
    if not isfile or not readfile or not isfile(Config.LocationFile) then
        return
    end

    local success, data = pcall(function()
        return HttpService:JSONDecode(readfile(Config.LocationFile))
    end)

    if not success or type(data) ~= "table" then return end

    if type(data.tabs) == "table" then
        Tabs = data.tabs
    end

    if type(data.locations) == "table" then
        Locations = data.locations
    elseif #data > 0 then
        Locations["Tab 1"] = data
    end

    ActiveTab = data.activeTab or Tabs[1] or "Tab 1"

    if #Tabs == 0 then
        Tabs = {"Tab 1"}
    end

    if not table.find(Tabs, ActiveTab) then
        ActiveTab = Tabs[1]
    end
end

local function saveLocations()
    if not writefile then return end

    pcall(function()
        writefile(Config.LocationFile, HttpService:JSONEncode({
            version = 2,
            tabs = Tabs,
            locations = Locations,
            activeTab = ActiveTab
        }))
    end)
end
